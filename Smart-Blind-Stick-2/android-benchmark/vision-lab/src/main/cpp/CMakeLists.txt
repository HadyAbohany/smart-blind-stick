cmake_minimum_required(VERSION 3.22)
project(sbs_qwen LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    # Debug APKs retain Java diagnostics, but tensor kernels must be optimized.
    add_compile_options(-O3)
    add_compile_definitions(NDEBUG)
endif()
if(NOT EXISTS "${SBS_LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Set SBS_LLAMA_SOURCE to the pinned llama.cpp source prepared by tools/prepare_vision_lab.py")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory("${SBS_LLAMA_SOURCE}" llama-build EXCLUDE_FROM_ALL)
if(ANDROID)
    add_library(sbs_qwen SHARED qwen_core.cpp qwen_jni.cpp)
    target_link_libraries(sbs_qwen PRIVATE mtmd llama android log)
    target_link_options(sbs_qwen PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_executable(sbs_qwen_smoke qwen_core.cpp qwen_smoke.cpp)
    target_link_libraries(sbs_qwen_smoke PRIVATE mtmd llama)
endif()
